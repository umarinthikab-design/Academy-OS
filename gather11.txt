import { prisma } from "./prisma";

// Append a row to the activity log. Called from every mutating server
// action so there's a full audit trail of who did what. `action` strings
// are kept consistent and snake_case, e.g. "created_drill",
// "approved_request" — matches the example comment on the ActivityLog
// model in prisma/schema.prisma.
export async function logActivity(
  userId: string,
  action: string,
  entityType: string,
  entityId?: string,
  details?: string
) {
  try {
    await prisma.activityLog.create({
      data: { userId, action, entityType, entityId, details },
    });
  } catch {
    // Logging must never break the action that triggered it. A failed log
    // write is swallowed rather than bubbling up as a 500.
  }
}

// Navigation is role + permission aware. The dashboard layout resolves which
// links the logged-in user can actually reach (via getPermissions) and passes
// the filtered list to both the desktop sidebar and mobile drawer — the same
// source for both, so they never drift.

import type { Permissions } from "./permissions";

export type NavItem = { href: string; label: string; icon: string; badge?: number };

export function getNavItems(perms: Permissions, requestsBadge = 0, registrationsBadge = 0, broadcastsBadge = 0): NavItem[] {
  const items: NavItem[] = [{ href: "/", label: "Dashboard", icon: "dashboard" }];
  items.push({ href: "/broadcasts", label: "Broadcasts", icon: "bell", badge: broadcastsBadge > 0 ? Math.min(broadcastsBadge, 9) : undefined });

  if (perms.role === "PARENT") {
    items.push({ href: "/settings", label: "Settings", icon: "settings" });
    return items;
  }

  // Requests are visible to coaching staff and admins alike: approvers action
  // pending proposals, requesters track their own submissions. The badge
  // highlights how many pending requests matter to this user. Assistant
  // coaches only ever submit requests, so their copy reads "Approvals"
  // (they're tracking the status of their submissions, not fielding them);
  // admins and head coaches keep "Requests" since they're approving.
  if (perms.isAdmin || perms.isClubManager || perms.isHeadCoach || perms.isAssistant) {
    items.push({ href: "/requests", label: perms.isAssistant ? "Approvals" : "Requests", icon: "inbox", badge: requestsBadge || undefined });
  }

  // Coaching staff (head + assistant) and admins share the football-ops pages.
  items.push(
    { href: "/schedule", label: "Schedule", icon: "calendar" },
    { href: "/sessions", label: "Session Plans", icon: "sessions" },
    { href: "/attendance", label: "Attendance", icon: "attendance" }
  );

  if (perms.isAdmin || perms.isClubManager || perms.isHeadCoach || perms.isAssistant) {
    items.push({ href: "/squad", label: "Squad", icon: "squad" });
  }

  // Roster & structure pages: admins and club managers always; head coaches
  // when they hold the matching permission. Assistants stay on the
  // player-facing tools.
  if (perms.isAdmin || perms.isClubManager || (perms.isHeadCoach && perms.canEditRoster)) {
    items.push({ href: "/coaches", label: "Coaches", icon: "coaches" });
  }
  if (perms.isAdmin || perms.isClubManager || perms.isHeadCoach) {
    items.push({ href: "/batches", label: "Batches", icon: "batches" });
  }
  if (perms.isAdmin || perms.isClubManager || (perms.isHeadCoach && perms.canEditLocations)) {
    items.push({ href: "/locations", label: "Locations", icon: "locations" });
  }
  if (perms.isAdmin || perms.isClubManager || (perms.isHeadCoach && perms.canEditAgeGroups)) {
    items.push({ href: "/age-groups", label: "Age Groups", icon: "target" });
  }

  if (perms.canSuggestDrills) {
    items.push({ href: "/drills", label: "Drill Library", icon: "drills" });
  }

  // Registration review - admin/club-manager only, matching every other
  // admin-management page built since Club Manager was introduced.
  if (perms.isAdmin || perms.isClubManager) {
    items.push({ href: "/registrations", label: "Registrations", icon: "squad", badge: registrationsBadge || undefined });
  }

  // Club-wide (academy) configuration - admin and club manager only.
  // Distinct from the personal /settings page every user gets.
  if (perms.isAdmin || perms.isClubManager) {
    items.push({ href: "/academy-settings", label: "Club Settings", icon: "settings" });
  }

  items.push({ href: "/activity", label: "Activity", icon: "activity" });
  items.push({ href: "/settings", label: "Settings", icon: "settings" });

  return items;
}

// ---- Grouping ----
// Items keep the order and permission filtering from getNavItems; this only
// buckets them under headings. Headings are skipped for short lists (parents),
// and a group with no visible items never appears.

export type NavSection = { label: string | null; items: NavItem[] };

const NAV_GROUPS: { label: string; hrefs: string[] }[] = [
  { label: "Overview", hrefs: ["/", "/broadcasts", "/requests"] },
  { label: "Training", hrefs: ["/schedule", "/sessions", "/attendance", "/drills"] },
  { label: "People", hrefs: ["/squad", "/coaches", "/registrations"] },
  { label: "Organisation", hrefs: ["/batches", "/age-groups", "/locations"] },
  { label: "System", hrefs: ["/academy-settings", "/activity", "/settings"] },
];

export function groupNavItems(items: NavItem[]): NavSection[] {
  if (items.length < 5) return [{ label: null, items }];
  const sections: NavSection[] = NAV_GROUPS.map((g) => ({ label: g.label, items: [] as NavItem[] }));
  const fallback = sections[sections.length - 1];
  for (const item of items) {
    const idx = NAV_GROUPS.findIndex((g) => g.hrefs.includes(item.href));
    (idx === -1 ? fallback : sections[idx]).items.push(item);
  }
  return sections.filter((s) => s.items.length > 0);
}

model ActivityLog {
  id         String   @id @default(cuid())
  userId     String
  user       User     @relation(fields: [userId], references: [id])
  action     String // e.g. "created_drill", "approved_request"
  entityType String // e.g. "Drill", "ScheduledSession"
  entityId   String?
  details    String?
  createdAt  DateTime @default(now())
}

enum Role {
  ADMIN
  CLUB_MANAGER
  HEAD_COACH
  ASSISTANT_COACH
  PARENT
}

model User {
  id             String   @id @default(cuid())
  email          String   @unique
  password       String // hashed - never store plain text
  name           String
  role           Role
  // Bump this to invalidate every existing session for this user without
  // rotating SESSION_SECRET (which would log everyone out). The value is
  // baked into the JWT at login time and checked against this column on
  // every request in getSession().
  sessionVersion Int      @default(0)
  createdAt      DateTime @default(now())

  // Profile photo (URL for now - same approach as Player.photoUrl). Optional
  // so admins and anyone without a hosted avatar can skip it.
  photoUrl String?

  // UI theme preference. Persisted per user so the whole app renders in the
  // same mode across devices.
  theme String @default("light")

  // When true the user must change their password before using the app.
  // Set when an admin creates the account or resets the password; cleared
  // once they complete a password change.
  mustChangePassword Boolean @default(false)

  // Set when a club manager / admin archives this staff member. Null =
  // active. Moved from Coach so it covers users with no Coach row (admins,
  // club managers) too. Historical data (drills, sessions, attendance) is
  // never touched by archiving - only login access and active-selection
  // visibility.
  archivedAt DateTime?

  coach        Coach?
  parent       Parent?
  activityLogs ActivityLog[]
  approvalMessages ApprovalMessage[]
  pushSubscriptions PushSubscription[]
  reviewedRegistrations PendingRegistration[]
  notifications Notification[]
}
