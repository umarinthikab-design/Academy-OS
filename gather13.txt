type SquadPosition = { url: string; playerId: string; y: number; expanded: string[]; restore?: boolean };
function readPosition(): SquadPosition | null {
  try {
    const value = JSON.parse(sessionStorage.getItem(SQUAD_RETURN_KEY) ?? "null");
    return value && typeof value.url === "string" && typeof value.playerId === "string" && Number.isFinite(value.y) ? value : null;
  } catch { return null; }
}
function writePosition(value: SquadPosition) {
  try { sessionStorage.setItem(SQUAD_RETURN_KEY, JSON.stringify(value)); } catch { /* Anchor navigation remains available. */ }
}

export function SquadPlayerLink({ playerId, appearance = "button", children }: { playerId: string; appearance?: "button" | "name"; children: ReactNode }) {
  const pathname = usePathname();
  const params = useSearchParams();
  const query = params.toString();
  const returnTo = safeSquadReturn(`${pathname}${query ? "?" + query : ""}#player-card-${playerId}`);
  const href = `/squad/${playerId}?returnTo=${encodeURIComponent(returnTo)}`;
  function remember() {
    const expanded = Array.from(document.querySelectorAll(".squad-grid > div"))
      .filter(card => card.querySelector("details[open]")) .map(card => card.id);
    writePosition({ url: returnTo, playerId, y: window.scrollY, expanded });
  }
  return <Link href={href} onClick={remember} className={appearance === "button" ? "ui-btn" : "text-link"}
    style={appearance === "button" ? { display: "inline-flex", padding: "5px 12px", fontSize: 13, fontWeight: 700, border: "1px solid var(--border)", borderRadius: 8, color: "var(--text-heading)", background: "var(--surface)", textDecoration: "none" } : { fontSize: 16, fontWeight: 800, color: "var(--text-heading)", textDecoration: "none" }}>{children}</Link>;
}

export function BackToSquad({ playerId, returnTo }: { playerId: string; returnTo: string }) {

    const saved = readPosition();
    if (saved?.playerId === playerId && returnTo === "/squad") setDestination(safeSquadReturn(saved.url));

}

"use client";

import { useEffect, useState, type ReactNode } from "react";
import { Button } from "./Button";

// Reusable view preference: storage is read after hydration and scoped per page.
export function ListView({ storageKey, children }: { storageKey: string; children: ReactNode }) {
  const [view, setView] = useState<"tiles" | "list">("tiles");
  useEffect(() => {
    try { if (localStorage.getItem(storageKey) === "list") setView("list"); } catch { /* Storage may be blocked. */ }
  }, [storageKey]);
  function choose(next: "tiles" | "list") {
    setView(next);
    try { localStorage.setItem(storageKey, next); } catch { /* Preference still works for this visit. */ }
  }
  return (
    <div>
      <div className="view-switch" role="group" aria-label="Session display">
        <span style={{ fontSize: 13, color: "var(--text-muted)" }}>View:</span>
        <Button type="button" size="sm" variant={view === "tiles" ? "approval" : "outline"} aria-pressed={view === "tiles"} onClick={() => choose("tiles")}>Tiles</Button>
        <Button type="button" size="sm" variant={view === "list" ? "approval" : "outline"} aria-pressed={view === "list"} onClick={() => choose("list")}>List</Button>
      </div>
      <div className={view === "tiles" ? "compact-grid" : undefined} style={view === "list" ? { display: "flex", flexDirection: "column", gap: 10 } : undefined}>{children}</div>
    </div>
  );
}

        <h3 style={{ fontFamily: "var(--font-headline)", fontSize: 24, fontWeight: 600, color: "var(--text-heading)" }}>
          Needs review
        </h3>
        <span style={{ fontSize: 13, color: "var(--text-faint)" }}>({sessions.length})</span>

        <ListView storageKey="touchline-attendance-view">
          {sessions.map((s) => {

            const started = coachId ? opened : false;

            return (

                              {p.status !== "ACTIVE" && (

                              {marked.has(p.id) && <span style={{ fontSize: 13, color: "var(--success)", fontWeight: 700 }}>âœ“ recorded</span>}

                      ))}
                    </div>
                    <Button full style={{ marginTop: 14, padding: "12px 18px", fontSize: 14 }}>
                      {isAdmin ? "Record attendance" : "Submit for approval"}
                    </Button>
                    <p style={{ fontSize: 13, color: "var(--text-faint)", marginTop: 8 }}>
                      {isAdmin
                        ? "Records directly â€” no approval needed."

                    </p>
                  </form>

              </details>
            );
          })}
        </ListView>
      )}

      {/* â”€â”€ Rollup â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */}

        <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", gap: 10, flexWrap: "wrap" }}>
          <h3 style={{ fontFamily: "var(--font-headline)", fontSize: 24, fontWeight: 600, color: "var(--text-heading)", margin: 0 }}>Rollup</h3>
          <form style={{ display: "flex", gap: 8, alignItems: "center" }}>
